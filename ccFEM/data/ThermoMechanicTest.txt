[materials]		
	[./matrixmaterial]					 
		E				=	3e4
		v				=	0.3
		lengthscale		=	0.05	#0.5
		MaxStrain		=	0.01 
		DamageBeta		=	100
		CTE             =       5.0e-6      #1        
		k               =       10.0         #mJ/mm/K e0
		#Q_hs           =       7.0e-2       #mW/mm^3 e-6
		rho             =       5.0e-9     #t/mm^3  e-12
		T_ref           =       300			#K
		T_init          =       300
		C_p             =       280e6       #mJ/t/K e6
	[../crackmaterial]					 
		E				=	3e4
		v				=	0.3
		lengthscale		=	0.05	#0.5
		MaxStrain		=	0.01

		CTE             =       5.0e-6      #1        
		k               =       10.0          #mJ/mm/K e0
		#Q_hs           =       7.0e-2       #mW/mm^3 e-6
		rho             =       5.0e-9     #t/mm^3  e-12
		T_ref           =       300			#K
		T_init          =       300
		C_p             =       280e6       #mJ/t/K e6
		MaterialDamage	=	1.0

		elementsetid	=	0		#start from 0 
		material		=	matrixmaterial
	[../crack]
		elementsetid	=	1		#start from 0 
		material		=	crackmaterial

	[./botEdge]
		elementsetid	=	0		#start from 0 
		type			=	displacement 
		ux				=	0.0
		uy				=	0.0
	[../topEdge]
		elementsetid	=	1

[]

	[./botEdge]
			elementsetid    = 0
        	type            = temperature		# 第一类边界条件

	[../topEdge]
			elementsetid    = 1
        	type            = temperature		# 第一类边界条件
        	T         		= 800    				# 
[]

	#reorderingreuse		=	false
